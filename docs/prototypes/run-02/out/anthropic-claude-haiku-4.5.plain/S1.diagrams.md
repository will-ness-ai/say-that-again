```text
on(/pillar-visibility channel:#x visibility:public|private)
  if declared state matches current state
    report convergence
    return
  
  if visibility:public
    count = verified members without composite role
    prompt "This makes full history readable by ~{count}. A later close does not undo that. Continue?"
    on(Confirm)
      grant read_message_history
      report success
    on(Cancel or 60s timeout)
      report cancelled
  
  if visibility:private
    count = verified members without composite role
    prompt "~{count} verified members lose access to #x. They get no notification. Continue?"
    on(Confirm)
      revoke read_message_history
      report success
    on(Cancel or 60s timeout)
      report cancelled

on(/pillar-visibility channel:all visibility:public|private)
  channels = all pillar channels
  count = sum of members affected across all channels
  prompt "Batch: {count} members affected across {len(channels)} channels. Continue?"
  on(Confirm)
    for each channel
      apply visibility change
      repair drift if found
    report batch complete
  on(Cancel or 60s timeout)
    report cancelled
```
