on(/pillar-visibility channel:#x visibility:public|private)
  if declared state matches current state
    report converged, repair any drift
    return
  
  if visibility:public
    count = verified members − composite members
    prompt "History readable by ~{count}. Later close won't undo. Continue?"
    on Confirm → open channel, report
    on Cancel or expire(60s) → return
  
  if visibility:private
    count = verified members − composite members
    prompt "~{count} members lose access, no notification. Continue?"
    on Confirm → close channel, report
    on Cancel or expire(60s) → return

on(/pillar-visibility channel:all visibility:public|private)
  collect all pillar channels needing transition
  if none
    report all converged
    return
  
  count = (verified − composite) × channels needing transition
  prompt "{count} total access changes across {N} channels. Continue?"
  on Confirm → apply all, report per-channel
