on(/pillar-visibility)
  if declared state matches current state
    report convergence
    return
  
  if direction is open (→ public)
    count = verified members − composite members
    prompt: "Makes history readable by ~{count}. A later close does not undo. Continue?"
    on(Confirm) → grant read_message_history, report
    on(Cancel or 60s timeout) → report cancellation
  
  if direction is close (→ private)
    count = verified members − composite members
    prompt: "~{count} members lose access. No notification sent. Continue?"
    on(Confirm) → revoke read_message_history, report
    on(Cancel or 60s timeout) → report cancellation
  
  if batch form (channel: all)
    count = sum of all transitions across every pillar channel
    prompt: "Affects ~{count} members across {N} channels. Continue?"
    on(Confirm) → apply to all, report per-channel
    on(Cancel or 60s timeout) → report cancellation
